\begin{algorithm}[t]
  \caption{Term-type extraction for a term $t$. \todo{finalise}}
  \label{alg:extract}
  \KwIn{term $t$, corpus $\mathcal{D}$, instructions $I$}
  \KwOut{typed definitions $\mathcal{Y}_t$ with provenance}
  $\mathcal{Y}_t \gets \emptyset$\;
  $a \gets \textsc{LLM}(I \,\|\, \mathcal{D} \,\|\, q(t))$ \tcp*{$\mathcal{D}$ served from cache}
  \ForEach{$(y, \text{cites}) \in \textsc{Parse}(a)$}{
    \If{$\textsc{Verify}(\text{cites}, \mathcal{D})$}{
      $\mathcal{Y}_t \gets \mathcal{Y}_t \cup \set{(y, \text{cites})}$\;
    }
  }
  \Return $\mathcal{Y}_t$\;
\end{algorithm}
