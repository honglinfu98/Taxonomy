\section{Citation Verification}
\label{app:verify}

\begin{algorithm}[t]
  \caption{Verify one citation $(i, \ell, q)$ against corpus $\mathcal{D}$.}
  \label{alg:verify}
  \KwIn{document index $i$, section label $\ell$, quote $q$; shingle length $k{=}12$, threshold $\tau{=}0.8$}
  \KwOut{status $\in$ \{ok, relabelled, rejected\}, possibly corrected $\ell$}
  \If{$d_i \notin \mathcal{D}$}{\Return rejected\;}
  \If{$\ell \in d_i$ \textbf{and} \textsc{Found}$(q, s_{i\ell})$}{\Return ok\;}
  \ForEach{$\ell' \in d_i$}{
    \If{\textsc{Found}$(q, s_{i\ell'})$}{$\ell \gets \ell'$; \Return relabelled\;}
  }
  \Return rejected\;
  \BlankLine
  \SetKwFunction{Found}{Found}
  \SetKwProg{Fn}{Function}{:}{}
  \Fn{\Found{$q$, $s$}}{
    $q, s \gets \textsc{Norm}(q), \textsc{Norm}(s)$ \tcp*{lower-case alphanumerics}
    \lIf{$q \subseteq s$}{\Return true}
    $G \gets$ $k$-shingles of $q$; $H \gets$ positions of each $g \in G$ in $s$\;
    \Return $\max_{\text{window } w \subseteq s,\ |w| = 2|q|} |\{g \in G : g \text{ occurs in } w\}| / |G| \ge \tau$\;
  }
\end{algorithm}

\Cref{alg:verify} is the procedure of \cref{sec:method:verify}. Normalisation
removes everything but lower-case letters and digits, so line breaks,
hyphenation and the spacing artefacts of PDF extraction do not matter. The
window constraint is what distinguishes a compressed quote, whose shingles
cluster in one place, from a paraphrase assembled from phrases that occur far
apart.
